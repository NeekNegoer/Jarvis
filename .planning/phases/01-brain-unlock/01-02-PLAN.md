      cmd = self._build_cmd(user_text)
      idx = cmd.index("--output-format")
      cmd[idx + 1] = "stream-json"

    Add `think_stream(self, user_text)` as a generator (do NOT remove or alter think()):
      - cmd = self._build_stream_cmd(user_text)
      - timeout = COMPUTER_TIMEOUT if self.computer_use else CHAT_TIMEOUT
      - proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
      - final_text = ""
      - for raw_line in proc.stdout: strip; skip empty; try json.loads, on JSONDecodeError continue.
        Use obj.get(...) at every level (defensive — tool shapes are MEDIUM confidence).
        t = obj.get("type","")
          t == "assistant": for block in (obj.get("message") or {}).get("content", []):
              bt = block.get("type","")
              bt == "thinking": continue
              bt == "text": text = block.get("text","");  if text: yield {"type":"text","text":text}
              bt == "tool_use": yield {"type":"tool_use","name":block.get("name"),"input":block.get("input",{})}
          t == "tool_result": for block in obj.get("content", []):
              if block.get("type") == "image":
                  yield {"type":"screenshot","data": block.get("source",{}).get("data","")}
          t == "result": final_text = (obj.get("result") or "").strip()
              if obj.get("session_id"): self.session_id = obj["session_id"]
              yield {"type":"result","text": final_text or "Okay."}
          # skip type in {"system","rate_limit_event"} — no user-facing content
      - finally: try proc.wait(timeout=5) except subprocess.TimeoutExpired: proc.kill()
      - After the loop: if proc.returncode and not final_text:
          stderr = proc.stderr.read()
          if any(w in stderr.lower() for w in ("network","connect","timeout","offline","fetch failed")):
              raise BrainOffline(stderr.strip())
          yield {"type":"error","text": stderr[:200] or "Something went wrong."}
      - Always finish with: yield {"type":"done"}

    Do not import any non-stdlib module. Keep Python 3.9 compatible (no match statements, no new syntax).
